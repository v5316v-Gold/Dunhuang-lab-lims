# 环境变量
BACKUP_DIR=/var/backups/dunhuang-lims
POSTGRES_PASSWORD=__FROM_ENV_FILE__
GPG_RECIPIENT=lims-backup@dunhuang-lab.local
WAL_CONTAINER=dunhuang-pg-prod
PG_CONTAINER=dunhuang-pg-prod

# 每日 02:00 全量备份
0 2 * * * /opt/dunhuang-lab-lims/deploy/backup/pg-full-backup.sh >> /var/log/lims-backup.log 2>&1

# 每 15 分钟 WAL 归档
*/15 * * * * /opt/dunhuang-lab-lims/deploy/backup/wal-archive.sh >> /var/log/lims-wal.log 2>&1

# 每日 04:00 备份健康检查
0 4 * * * /opt/dunhuang-lab-lims/deploy/backup/verify-backup.sh >> /var/log/lims-backup-check.log 2>&1

# 每月 1 日 06:00 还原演练(测试容器)
0 6 1 * * /opt/dunhuang-lab-lims/deploy/backup/restore.sh $(find $BACKUP_DIR -name "pg_full_*.dump*" -mtime -2 | sort | tail -1) >> /var/log/lims-restore-test.log 2>&1
